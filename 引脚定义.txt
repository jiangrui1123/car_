#ifndef __BOARD_H__
#define __BOARD_H__

#include <REGX52.H>

/*==============================================================================
 * car_simple —— STC89C52RC 引脚分配表（全工程唯一来源）
 *
 *  ┌──────┬───────────────────┬──────┬──────────────────────────────┐
 *  │ 引脚 │ 信号名            │ 方向 │ 外接                          │
 *  ├──────┼───────────────────┼──────┼──────────────────────────────┤
 *  │ P1.0 │ motor_left_PWM    │ 输出 │ L298N ENB   左电机 PWM        │
 *  │ P1.1 │ Left_IN1          │ 输出 │ L298N IN3   左电机方向        │
 *  │ P1.2 │ Left_IN2          │ 输出 │ L298N IN4   左电机方向        │
 *  │ P1.3 │ Right_IN1         │ 输出 │ L298N IN1   右电机方向        │
 *  │ P1.4 │ Right_IN2         │ 输出 │ L298N IN2   右电机方向        │
 *  │ P1.5 │ motor_right_PWM   │ 输出 │ L298N ENA   右电机 PWM        │
 *  ├──────┼───────────────────┼──────┼──────────────────────────────┤
 *  │ P2.0 │ OLED_SDA          │ 双向 │ OLED SDA（4.7k 上拉）         │
 *  │ P2.1 │ OLED_SCL          │ 输出 │ OLED SCL（4.7k 上拉）         │
 *  │ P2.2 │ Buzzer            │ 输出 │ 蜂鸣器模块 I/O                │
 *  │ P2.3 │ tracking_left_2   │ 输入 │ 循迹探头 最左                 │
 *  │ P2.4 │ tracking_left_1   │ 输入 │ 循迹探头 左                   │
 *  │ P2.5 │ tracking_middle   │ 输入 │ 循迹探头 中                   │
 *  │ P2.6 │ tracking_right_1  │ 输入 │ 循迹探头 右                   │
 *  │ P2.7 │ tracking_right_2  │ 输入 │ 循迹探头 最右                 │
 *  ├──────┼───────────────────┼──────┼──────────────────────────────┤
 *  │ P3.0 │ RXD               │ 输入 │ 蓝牙模块 TXD（1k + 跳线）     │
 *  │ P3.1 │ TXD               │ 输出 │ 蓝牙模块 RXD（1k + 跳线）     │
 *  │ P3.2 │ （空闲）          │  —   │ INT0，可留给编码器 A          │
 *  │ P3.3 │ （空闲）          │  —   │ INT1，可留给编码器 B          │
 *  │ P3.4 │ （空闲）          │  —   │ T0，可留给计数器              │
 *  │ P3.5 │ （空闲）          │  —   │ T1，可留给计数器              │
 *  │ P3.6 │ （空闲）          │  —   │                              │
 *  │ P3.7 │ LED               │ 输出 │ 指示灯（串 1k）               │
 *  └──────┴───────────────────┴──────┴──────────────────────────────┘
 *
 *  空闲：P1.6  P1.7  P3.2 ~ P3.6
 *============================================================================*/

/* ---------------- 电机（L298N） ---------------- */
sbit motor_left_PWM  = P1^0;
sbit motor_right_PWM = P1^5;
sbit Left_IN1        = P1^1;
sbit Left_IN2        = P1^2;
sbit Right_IN1       = P1^3;
sbit Right_IN2       = P1^4;

/* ---------------- OLED / 蜂鸣器 / 指示灯 ---------------- */
sbit OLED_SCL = P2^1;
sbit OLED_SDA = P2^0;
sbit Buzzer   = P2^2;
sbit LED      = P3^7;

/* ---------------- 五路循迹探头 ---------------- */
sbit tracking_left_2  = P2^3;
sbit tracking_left_1  = P2^4;
sbit tracking_middle  = P2^5;
sbit tracking_right_1 = P2^6;
sbit tracking_right_2 = P2^7;

/* 一次读回 5 位图案：bit4..bit0 = L2 L1 M R1 R2，0 = 黑，1 = 白 */
#define TRACK_PATTERN()   ((P2 >> 3) & 0x1F)

/* 全黑（横线 / 宽黑区）：5 路全为 0 */
#define TRACK_ALL_BLACK() (((P2 >> 3) & 0x1F) == 0x00)

/*==============================================================================
 * 注意事项
 *============================================================================*/

/*  【1】传感器电平：0 = 黑线，1 = 白底
 *      代码里写的是 tracking_middle == 0 才算"检测到黑线"。
 *      如果你换了一个"1 = 黑"的模块，所有判定会整体反转，
 *      现象是：车看到白底就转、看到黑线就直冲。
 *      接线前先用万用表或 OLED 显示的图案核对一次。
 *
 *  【2】P2 口一分为三：I2C（P2.0/P2.1）+ 蜂鸣器（P2.2）+ 5 路输入（P2.3~P2.7）
 *      只能逐位操作（sbit 赋值），绝对不要出现  P2 = 0xNN;  这种整字节写。
 *      一旦写了整字节，5 路探头的输入锁存会被覆盖，循迹立刻失效。
 *      本工程所有访问都用 sbit，安全；以后加代码时务必遵守。
 *
 *  【3】蜂鸣器（P2.2）紧挨 I2C（P2.1）
 *      有源蜂鸣器是感性负载，开关瞬间会产生尖峰。若出现偶发花屏：
 *        a) 蜂鸣器改用三极管驱动，基极串 1k；
 *        b) 蜂鸣器两端反并一只 1N4148；
 *        c) 实在不行把蜂鸣器挪到 P1.6 或 P1.7。
 *
 *  【4】P3.0 / P3.1 既是串口、也是 STC-ISP 下载口
 *      蓝牙模块和下载器会同时驱动 P3.0，导致"有时能下载、有时握手失败"。
 *      务必：蓝牙 TX/RX 各串 1k 电阻，并留一个 2 针跳线，下载时拔掉。
 *
 *  【5】L298N 的 ENA / ENB 跳线帽必须拔掉
 *      不拔的话 ENA/ENB 被硬拉到 5V，PWM 完全不起作用，车永远是全速。
 *      这是"调速无效"最常见的原因。
 *
 *  【6】电机方向真值表（与本工程代码一致）
 *      IN1  IN2   状态
 *       0    1    前进
 *       1    0    后退
 *       1    1    短路制动（motor_brake）
 *       0    0    滑行（motor_stop）
 *      若某个轮子转向相反，交换该电机两根输出线，不要改代码。
 *
 *  【7】PWM 极性：高电平使能
 *      中断里写的是  motor_left_PWM = (pwm < duty_L);
 *      duty_L 范围 0~50，对应占空比 0~100%（motor_speed 里已经除 2）。
 *      直接把 duty_L 赋 100 会变成"常开"，调速失效。
 *
 *  【8】OLED 上拉电阻
 *      P2.0/P2.1 是准双向口，模拟 I2C 必须外接 4.7k 上拉。
 *      模块 3.3V 供电时上拉到 3.3V，不要拉到 5V。
 *      没有上拉的现象：OLED 全黑、偶尔闪一下、或指针停在初始化。
 *
 *  【9】共地
 *      单片机 5V、L298N 的 12V 电池、5 路探头、蓝牙模块必须共地，
 *      否则 PWM 电平没有参考，电机行为会漂移，OLED 也可能通信失败。
 *
 *  【10】电源去耦
 *      L298N 电源端并 470~1000uF；单片机电源入口 10uF + 100nF；
 *      电机启动/反转瞬间电流可达数安，电池内阻会造成 5V 跌落，
 *      典型现象是"一拐弯就重启"。
 *
 *  【11】晶振与复位
 *      11.0592MHz + 两只 22~30pF；复位用 10uF + 10k + 按键；EA 接 VCC。
 *      波特率 9600 是按 11.0592MHz 算的，换成 12MHz 会有 8.5% 误差。
 *
 *  【12】空闲引脚
 *      P1.6 / P1.7 / P3.2~P3.6 未使用，其中 P3.2/P3.3 是 INT0/INT1、
 *      P3.4/P3.5 是 T0/T1。以后要加编码器或按键，优先用这四个。
 *
 *============================================================================*/

#endif
